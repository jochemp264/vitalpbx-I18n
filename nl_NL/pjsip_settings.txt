pjsip_settings.mwi_settings = MWI-instellingen
pjsip_settings.mwi_tps_queue_high = Taskprocessor-wachtrij hoog
pjsip_settings.mwi_tps_queue_high.tooltip =
	Op een zwaar beladen systeem moet u mogelijk de limieten van de taskprocessor-wachtrij aanpassen. Als een taskprocessor-wachtrij
	het hoge waterniveau bereikt, stopt pjsip met het verwerken van nieuwe verzoeken totdat de waarschuwing is opgeheven. De waarschuwing
	wordt opgeheven wanneer alle waarschuwende taskprocessor-wachtrijen tot hun lage waterniveau zijn gedaald.
pjsip_settings.mwi_tps_queue_low = Taskprocessor-wachtrij laag
pjsip_settings.mwi_tps_queue_low.tooltip =
	Op een zwaar beladen systeem moet u mogelijk de limieten van de taskprocessor-wachtrij aanpassen. Als een taskprocessor-wachtrij
	het hoge waterniveau bereikt, stopt pjsip met het verwerken van nieuwe verzoeken totdat de waarschuwing is opgeheven. De waarschuwing
	wordt opgeheven wanneer alle waarschuwende taskprocessor-wachtrijen tot hun lage waterniveau zijn gedaald. Stel in op -1 zodat het lage
	waterniveau 90% van het hoge waterniveau is.
pjsip_settings.mwi_disable_initial_unsolicited = Initiële MWI uitschakelen
pjsip_settings.mwi_disable_initial_unsolicited.tooltip =
	Wanneer de initiële ongevraagde MWI-meldingen bij opstarten zijn ingeschakeld, worden de initiële meldingen bij
	opstarten verzonden. Als u veel eindpunten (duizenden) hebt die ongevraagde MWI gebruiken, kunt u overwegen
	de initiële opstartmeldingen uit te schakelen. Wanneer de initiële ongevraagde MWI-meldingen bij opstarten zijn uitgeschakeld,
	starten de meldingen bij de volgende contactupdate van het eindpunt.

pjsip_settings.allow_wildcard_certs = Wildcardcertificaten toestaan
pjsip_settings.allow_wildcard_certs.tooltip =
    In combinatie met "Server verifiëren" staan wildcards toe indien ingeschakeld, d.w.z. '.' in certificaten voor common en subject
    alt names van type DNS voor TLS-transporttypes. Namen moeten met de wildcard beginnen. Gedeeltelijke wildcards, bijv. 'f.example.com'
    en 'foo..com' zijn niet toegestaan. Namen matchen ook slechts één niveau, wat betekent dat '.example.com' overeenkomt met 'foo.example.com',
    maar niet met 'foo.bar.example.com'.
